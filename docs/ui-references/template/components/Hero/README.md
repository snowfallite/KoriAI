# Hero

Первый экран: градиентное поле с вложенными арками-лесенками и лучами, логотип K Ï O K U на нижнем срезе и кана под ним.

- Арт генерируется скриптом в SVG (`#ki-hero-art`) и берёт цвета из токенов (`--pink`, `--red`, `--ground`) — в теме `day` поле растворяется в белом, а не в чёрном.
- Параметры скрипта: `ROW`/`BW` (размер блока), шаг радиусов арок, `RAYS` (число лучей), `SPRING` (линия, ниже которой арки идут вертикально).
- Логотип ставится отрицательным отступом на нижний край арта (`-mt-[0.52em]`), чтобы буквы «вырастали» из затухания.
- Арт — чистый декор: `aria-hidden="true"`; заголовок страницы — сам логотип в `<h1>`.

Не делайте: не кладите текст поверх середины градиента, не добавляйте второй градиент на экран.

## Разметка (HTML + Tailwind)

```html
<link rel="stylesheet" href="https://fonts.googleapis.com/css2?family=Noto+Sans+JP:wght@300&display=swap">

<section class="relative px-12 pt-12 pb-32" aria-label="Kïoku">
  <!-- art: generated by the script below; aria-hidden, pure decoration -->
  <svg id="ki-hero-art" class="block aspect-[1184/560] w-full" viewBox="0 0 1184 560" preserveAspectRatio="none" aria-hidden="true"></svg>

  <!-- wordmark sits on the art's bottom edge -->
  <h1 class="-mt-[0.52em] grid grid-cols-5 place-items-center px-[8%] text-[120px] leading-none font-normal">
    <span>K</span><span>Ï</span><span>O</span><span>K</span><span>U</span>
    <span></span>
    <span lang="ja" class="mt-6 font-kana text-word font-light text-ink-muted">き</span>
    <span lang="ja" class="mt-6 font-kana text-word font-light text-ink-muted">お</span>
    <span lang="ja" class="mt-6 font-kana text-word font-light text-ink-muted">く</span>
    <span></span>
  </h1>
</section>

<script>
/* Kïoku hero art — nested pixel arches + rays, faded into the page ground.
   Tunables: ROW (block height), BW (block width), arches (count / step), RAYS. */
(function () {
  var svg = document.getElementById('ki-hero-art'), NS = 'http://www.w3.org/2000/svg';
  var W = 1184, H = 560, CX = W / 2, SPRING = H * 0.42, ROW = 50, BW = 30, RAYS = 56;
  function el(tag, attrs, parent) {
    var n = document.createElementNS(NS, tag);
    for (var k in attrs) n.setAttribute(k, attrs[k]);
    (parent || svg).appendChild(n); return n;
  }
  var defs = el('defs', {});
  var bg = el('linearGradient', { id: 'ki-bg', x1: 0, y1: 0, x2: 0, y2: 1 }, defs);
  el('stop', { offset: '0', style: 'stop-color:var(--pink)' }, bg);
  el('stop', { offset: '0.3', style: 'stop-color:var(--pink)' }, bg);
  el('stop', { offset: '0.8', style: 'stop-color:var(--red)' }, bg);
  el('stop', { offset: '1', style: 'stop-color:var(--red)' }, bg);
  var fade = el('linearGradient', { id: 'ki-fade', x1: 0, y1: 0, x2: 0, y2: 1 }, defs);
  el('stop', { offset: '0.5', style: 'stop-color:var(--ground);stop-opacity:0' }, fade);
  el('stop', { offset: '0.97', style: 'stop-color:var(--ground);stop-opacity:1' }, fade);

  el('rect', { width: W, height: H, fill: 'url(#ki-bg)' });

  // rays from a focal point low in the centre
  var rays = el('g', { style: 'stroke:var(--red);stroke-width:1' });
  var FX = CX, FY = H * 0.9;
  for (var i = 0; i <= RAYS; i++) {
    var t = i / RAYS, x = -W * 0.15 + t * W * 1.3;
    el('line', { x1: FX, y1: FY, x2: x, y2: 0 }, rays);
  }
  for (var j = 1; j <= 6; j++) {
    el('line', { x1: FX, y1: FY, x2: 0, y2: j * H / 9 }, rays);
    el('line', { x1: FX, y1: FY, x2: W, y2: j * H / 9 }, rays);
  }

  // nested arches drawn as stepped blocks: circle above the springline, straight legs below
  var blocks = el('g', { style: 'fill:var(--red)' });
  for (var r = 96; r <= 640; r += 76) {
    for (var y = ROW / 2; y < H; y += ROW) {
      var dy = SPRING - y, dx;
      if (dy > 0) { if (r <= dy) continue; dx = Math.sqrt(r * r - dy * dy); } else dx = r;
      var xs = [CX - dx, CX + dx];
      for (var s = 0; s < 2; s++) {
        var bx = Math.round((xs[s] - BW / 2) / 2) * 2;
        if (bx < -BW || bx > W) continue;
        el('rect', { x: bx, y: y - ROW / 2, width: BW, height: ROW }, blocks);
      }
    }
  }
  el('rect', { width: W, height: H, fill: 'url(#ki-fade)' });
})();
</script>
```
